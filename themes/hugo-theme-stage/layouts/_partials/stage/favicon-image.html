{{- /* Draws the letter favicon. Context: the letter (string). Returns dict small (32px) and large (180px, apple-touch). */ -}} {{- $out := dict -}} {{- $base := resources.Get "og/favicon-base.png" -}} {{- $font := resources.Get "og/fonts/DelaGothicOne-Latin.ttf" -}} {{- $letter := upper (replaceRE `[\p{So}\p{Sk}\x{FE0F}\x{200D}]` "" (string .)) -}} {{- if and $base $font $letter -}} {{- $letter = substr $letter 0 1 -}} {{- $text := images.Text $letter (dict "font" $font "size" 112 "color" "#ffffff" "x" 92 "y" 22 "alignx" "center") -}} {{- $large := $base | images.Filter $text -}} {{- $large = $large.Process "png" -}} {{- $out = dict "large" $large "small" ($large.Resize "32x32 png") -}} {{- end -}} {{- return $out -}}

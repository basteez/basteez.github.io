{{- /* Ransom-note letters for the link-preview cards: images.Overlay puts a cut-paper tile (assets/og/tiles/-.png) under each letter, images.Text writes the letter on it. Deterministic, like collage.html: an LCG seeded with hash.FNV32a of the text. Params (dict): text, x, y (top left), size (tile height in px), font, schemes ("all" | "hero") Returns dict: filters (slice), width (px, of the whole run) */ -}} {{- $text := .text -}} {{- $size := .size -}} {{- $tw := int (mul $size 0.8) }}{{ $gap := int (mul $size 0.06) }}{{ $space := int (mul $size 0.36) -}} {{- $schemes := slice (slice "white" "#000000") (slice "red" "#ffffff") (slice "black" "#ffffff") (slice "white" "#e60012") -}} {{- if eq .schemes "hero" }}{{ $schemes = slice (slice "white" "#000000") (slice "red" "#ffffff") (slice "white" "#e60012") (slice "red" "#000000") }}{{ end -}} {{- $x := hash.FNV32a $text -}} {{- $cx := .x -}} {{- $filters := slice -}} {{- range $ch := split $text "" -}} {{- if eq $ch " " -}} {{- $cx = add $cx $space -}} {{- else -}} {{- $r := slice -}} {{- range seq 3 -}} {{- $x = mod (add (mul $x 1664525) 1013904223) 4294967296 -}} {{- $r = $r | append (div $x 4294968) -}} {{- end -}} {{- $s := index $schemes (mod (index $r 0) (len $schemes)) -}} {{- $tile := resources.Get (printf "og/tiles/%s-%d.png" (index $s 0) (mod (index $r 1) 4)) -}} {{- $dy := sub (div (mul (index $r 2) (mul $size 0.12)) 1000) (mul $size 0.06) | int -}} {{- with $tile -}} {{- $t := .Resize (printf "%dx%d" $tw $size) -}} {{- $filters = $filters | append (images.Overlay $t $cx (add $.y $dy)) -}} {{- end -}} {{- $filters = $filters | append (images.Text (upper $ch) (dict "font" $.font "size" (int (mul $size 0.7)) "color" (index $s 1) "x" (add $cx (div $tw 2)) "y" (add $.y $dy (int (mul $size 0.1))) "alignx" "center")) -}} {{- $cx = add $cx $tw $gap -}} {{- end -}} {{- end -}} {{- return dict "filters" $filters "width" (sub $cx .x) -}}

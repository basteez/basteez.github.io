{{- /* Link-preview card (1200x630) for OpenGraph/Twitter. Returns an image resource, or nil. Everything is drawn at build time on assets/og/card-base.png (the red stage and the black slab, see _source/og-cards.html), from the site's own settings: - posts: series, title, date and reading time, and the brand; - every other page: the hero name as ransom-note tiles, role, stack, and the brand. images.Text only wraps at the edge of the image, so the title is wrapped here, by an estimated glyph width, to keep it inside the slab (x 66..806). */ -}} {{- $cfg := partialCached "stage/config.html" . site.Language.Lang -}} {{- $base := resources.Get "og/card-base.png" -}} {{- $display := resources.Get "og/fonts/DelaGothicOne-Latin.ttf" -}} {{- $mono := resources.Get "og/fonts/CourierPrime-Bold.ttf" -}} {{- $card := false -}} {{- if and $base $display $mono -}} {{- /* the card fonts have no emoji or pictographs: drop them */ -}} {{- $clean := `[\p{So}\p{Sk}\x{FE0F}\x{200D}]` -}} {{- $x := 66 }}{{ $maxW := 740.0 -}} {{- $filters := slice -}} {{- /* the brand, bottom right */ -}} {{- $brand := replaceRE $clean "" $cfg.brand | strings.TrimSpace -}} {{- if $brand -}} {{- $probe := partial "stage/og-tiles.html" (dict "text" $brand "x" 0 "y" 0 "size" 64 "font" $display) -}} {{- $tiles := partial "stage/og-tiles.html" (dict "text" $brand "x" (sub 1152 $probe.width) "y" 524 "size" 64 "font" $display) -}} {{- $filters = $filters | append $tiles.filters -}} {{- end -}} {{- if and .IsPage (in site.Params.mainSections .Section) -}} {{- $title := replaceRE $clean "" .Title | strings.TrimSpace | replaceRE `\s+` " " -}} {{- $series := index (.GetTerms "series") 0 -}} {{- $top := 124 }}{{ if $series }}{{ $top = 162 }}{{ end }}{{ $bottom := 466 -}} {{- /* largest size whose wrapped title fits both the slab width and the height above the date line (Dela Gothic One averages ~0.66em per glyph on titles) */ -}} {{- $size := 40 }}{{ $lines := slice -}} {{- range $s := slice 66 58 52 46 40 -}} {{- if not $lines -}} {{- $cw := mul $s 0.66 -}} {{- $acc := slice }}{{ $cur := "" -}} {{- range $w := split $title " " -}} {{- $try := $w }}{{ if ne $cur "" }}{{ $try = printf "%s %s" $cur $w }}{{ end -}} {{- if and (ne $cur "") (gt (mul (strings.RuneCount $try) $cw) $maxW) -}} {{- $acc = $acc | append $cur }}{{ $cur = $w -}} {{- else -}} {{- $cur = $try -}} {{- end -}} {{- end -}} {{- $acc = $acc | append $cur -}} {{- $h := mul (len $acc) (int (mul $s 1.22)) -}} {{- if or (le $h (sub $bottom $top)) (eq $s 40) }}{{ $lines = $acc }}{{ $size = $s }}{{ end -}} {{- end -}} {{- end -}} {{- with $series -}} {{- $filters = $filters | append (images.Text (printf "%s · %s" (upper (T "series")) (upper (replaceRE $clean "" .LinkTitle))) (dict "font" $mono "size" 30 "color" "#ff3344" "x" $x "y" 104)) -}} {{- end -}} {{- $lh := int (mul $size 1.22) -}} {{- range $i, $l := $lines -}} {{- $filters = $filters | append (images.Text $l (dict "font" $display "size" $size "color" "#ffffff" "x" $x "y" (add $top (mul $i $lh)))) -}} {{- end -}} {{- $meta := .Date | time.Format $cfg.dateFormat | upper -}} {{- if $cfg.article.readingTime }}{{ $meta = printf "%s · %s" $meta (upper (T "minRead" (dict "Count" .ReadingTime))) }}{{ end -}} {{- $filters = $filters | append (images.Text $meta (dict "font" $mono "size" 30 "color" "#ffffff" "x" $x "y" 500)) -}} {{- else -}} {{- /* site card: the hero name as tiles, sized so the longest line fits the slab */ -}} {{- $names := slice }}{{ range $cfg.hero.name }}{{ with replaceRE $clean "" . | strings.TrimSpace }}{{ $names = $names | append . }}{{ end }}{{ end -}} {{- $names = first 3 $names -}} {{- $longest := 1 }}{{ range $names }}{{ $longest = math.Max $longest (strings.RuneCount .) }}{{ end -}} {{- $size := int (math.Min 104 (div 700.0 (mul $longest 0.86)) (div 300.0 (math.Max 1 (len $names)))) -}} {{- $y := 84 -}} {{- range $i, $n := $names -}} {{- $t := partial "stage/og-tiles.html" (dict "text" $n "x" (add $x (mul $i (int (mul $size 0.45)))) "y" $y "size" $size "font" $display "schemes" "hero") -}} {{- $filters = $filters | append $t.filters -}} {{- $y = add $y $size (int (mul $size 0.18)) -}} {{- end -}} {{- with replaceRE $clean "" (string $cfg.hero.role) | strings.TrimSpace | upper -}} {{- $rs := 30 -}} {{- $w := math.Min 760 (add (int (mul (strings.RuneCount .) $rs 0.86)) 44) -}} {{- with resources.Get "og/strip.png" }}{{ $filters = $filters | append (images.Overlay (.Resize (printf "%dx56" (int $w))) $x (add $y 14)) }}{{ end -}} {{- $filters = $filters | append (images.Text . (dict "font" $display "size" $rs "color" "#000000" "x" (add $x 22) "y" (add $y 24))) -}} {{- $y = add $y 86 -}} {{- end -}} {{- with $cfg.hero.stack -}} {{- $stack := slice }}{{ range . }}{{ $stack = $stack | append (upper (replaceRE $clean "" (string .))) }}{{ end -}} {{- $filters = $filters | append (images.Text (delimit $stack " · ") (dict "font" $mono "size" 26 "color" "#ffffff" "x" $x "y" (add $y 6) "linespacing" 8)) -}} {{- end -}} {{- end -}} {{- $card = $base | images.Filter $filters -}} {{- end -}} {{- return $card -}}

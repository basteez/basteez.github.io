{{- /* Returns the first parameter that is set, among several dotted paths under site.Params, or the default. "Set" means present and not null: false, 0 and "" are real values. Params (dict): paths []string, default any */ -}} {{- $val := .default -}} {{- $found := false -}} {{- range $path := .paths -}} {{- if not $found -}} {{- $cur := site.Params }}{{ $ok := true -}} {{- range split $path "." -}} {{- if and $ok (reflect.IsMap $cur) }}{{ $cur = index $cur (lower .) }}{{ else }}{{ $ok = false }}{{ end -}} {{- end -}} {{- if and $ok (ne $cur nil) }}{{ $val = $cur }}{{ $found = true }}{{ end -}} {{- end -}} {{- end -}} {{- return $val -}}

{{- /* Images. Page resources and assets/ images get their size (no layout shift) and, for JPEG/PNG/WebP wider than 480px, a srcset. Everything is lazy-loaded. A block image with a title becomes a figure. */ -}} {{- $u := urls.Parse .Destination -}} {{- $src := .Destination -}} {{- $w := 0 }}{{ $h := 0 }}{{ $srcset := "" -}} {{- if not $u.IsAbs -}} {{- with or (.Page.Resources.Get $u.Path) (resources.Get $u.Path) -}} {{- $src = .RelPermalink -}} {{- if and (eq .ResourceType "image") (ne (lower (path.Ext .Name)) ".svg") -}} {{- $w = .Width }}{{ $h = .Height -}} {{- if and (in (slice "jpeg" "png" "webp") .MediaType.SubType) (gt .Width 480) -}} {{- $res := . }}{{ $set := slice -}} {{- range slice 480 1024 -}} {{- if lt . $w }}{{ $set = $set | append (printf "%s %dw" ($res.Resize (printf "%dx" .)).RelPermalink .) }}{{ end -}} {{- end -}} {{- $srcset = delimit ($set | append (printf "%s %dw" .RelPermalink .Width)) ", " -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $img := printf "" $src .PlainText (cond (gt $w 0) (printf " width=\"%d\" height=\"%d\"" $w $h) "") (cond (ne $srcset "") (printf " srcset=%q sizes=\"(min-width: 800px) 760px, 100vw\"" $srcset) "") -}} {{- if and .IsBlock .Title -}} {{- else -}} {{ $img | safeHTML }} {{- end -}}
